Joining :
-------

INNER JOIN :-

# 1 #

SELECT user.name, product.name AS product_name
FROM product_order
INNER JOIN user ON product_order.u_id = user.u_id
INNER JOIN product ON product_order.p_id = product.p_id

# 2 #

SELECT user.name, product.name AS product_name, product.price
FROM product_order
INNER JOIN user ON product_order.u_id = user.u_id
INNER JOIN product ON product_order.p_id = product.p_id
WHERE product.price > 3000

# 3 #

SELECT user.name, product.name AS product_name, product.price, product_order.quantity, product_order.quantity*product.price AS total_price
FROM product_order
INNER JOIN user ON product_order.u_id = user.u_id
INNER JOIN product ON product_order.p_id = product.p_id
WHERE product.price > 3000
